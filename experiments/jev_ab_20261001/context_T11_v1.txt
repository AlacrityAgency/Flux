    def save_memory(self, title, content, enabled=True, ident=None):
        title, content = clean_text(title, 120), clean_text(content, 12000)
        if not title or not content:
            raise ValueError("عنوان و متن حافظه را وارد کنید.")
        with self.repo._lock, self.conn:
            if ident:
                if not self.conn.execute("SELECT 1 FROM memories WHERE id=?", (ident,)).fetchone():
                    raise KeyError(ident)
                self.conn.execute('''UPDATE memories SET title=?,content=?,enabled=?,
                    revision=revision+1,updated_at=? WHERE id=?''', (title, content, int(enabled), now(), ident))
            else:
                ident = "mem_" + uuid.uuid4().hex[:12]
                self.conn.execute("INSERT INTO memories VALUES (?,?,?,?,1,?,?)",
                                  (ident, title, content, int(enabled), now(), now()))
        return dict(self.conn.execute("SELECT * FROM memories WHERE id=?", (ident,)).fetchone())

def clean_text(value: str, limit: int = 20000) -> str:
    value = value.strip()
    if len(value) > limit:
        raise ValueError(f"متن باید کمتر از {limit} نویسه باشد.")
    if re.search(r"sk-[A-Za-z0-9_-]{16,}|-----BEGIN .*PRIVATE KEY-----", value):
        raise ValueError("کلید یا اطلاعات محرمانه را وارد گفت‌وگو و حافظه نکنید.")
    return value

## معماری کوچک‌تر برای یک کاربر

Python + PySide6 برای پنجره و موتور، SQLite برای حافظه/وضعیت/دفتر مصرف. رابط با worker داخلی و event queue کار می‌کند تا هنگام تماس مدل مسدود نشود. سرویس HTTP و هفت endpoint پیوست برای این دمو لازم نیستند؛ API داخلی به شکل interface پایتونی جدا می‌ماند تا بعداً wrapper HTTP اضافه شود. موتور یک instance فعال برای پوشهٔ داده دارد.

داده زیر `%LOCALAPPDATA%\FluxDemo\` مستقل از فایل برنامه ذخیره می‌شود. برنامه با shortcut و runtime همراه اجرا شود؛ نسخهٔ portable کافی است و نصاب حرفه‌ای، updater و سرویس پس‌زمینه نیاز نیست. با بستن برنامه، اجرای تازه متوقف و checkpoint ثبت می‌شود؛ بعد از بازکردن، کار نامعلوم بدون بررسی دوباره ارسال نمی‌شود. یک export/backup سادهٔ سازگار SQLite و فایل‌ها برای جلوگیری از وابستگی به یک پوشه کافی است.

## حافظه و دقت در حد دمو

SQLite محلی، فایل‌های متنی منتخب و جست‌وجوی متنی فارسی کافی‌اند. هر نکتهٔ حافظه منبع، نسخه و وضعیت پیشنهادی/تأییدشده/منسوخ دارد. فقط بخش مرتبط در سقف context بازیابی می‌شود؛ کل تاریخچه با هر درخواست ارسال نمی‌شود. نتیجهٔ خود Agent بدون تأیید کاربر به حقیقت دائمی حافظه تبدیل نمی‌شود. ویرایش یا حذف، نمایه و cache وابسته را هم تازه می‌کند.

دقت با معیار روشن، ارجاع به منبع، بررسی مستقل و علامت‌زدن موارد نامعلوم کنترل می‌شود. خروجی کد در این دمو «پیشنهاد کد/diff بررسی‌شده» است. اجرای خودکار کد و shell، دستکاری مستقیم مخزن و runner ایزوله از دمو خارج‌اند؛ تست اجرا نشده با برچسب not_run نشان داده می‌شود. اگر معیار Task نیازمند اجرای تست باشد، نتیجه بدون آن ACCEPT_CANDIDATE نمی‌شود. این محدودیت دامنه را کم می‌کند و نباید در UI پنهان شود.

### نتیجهٔ بررسی GapGPT

در بررسی ۶ سپتامبر ۲۰۲۶، ابزار وب محتوای صفحهٔ [مدیریت توکن](https://gapgpt.app/platform-v2/tokens) را دریافت نکرد. درخواست مستقیم بدون کلید به `https://api.gapgpt.app/v1/models` پاسخ HTTP 401 داد؛ این فقط نیاز به احراز هویت endpoint را نشان می‌دهد. سازگاری SDK، اعتبار کلید، دسترسی پنج مدل، JSON Schema و قیمت‌ها با این بررسی تأیید نشده‌اند. هیچ درخواست تولیدی با کلید ارسال نشده است.

Base URL و پنج شناسه و نرخ‌های پیوست، ورودی درخواستی کاربر هستند؛ در manifest با `verified=false` ثبت شوند. نام مشابه در ابزار توسعه اثبات دسترسی GapGPT نیست. جایگزینی پنهانی مدل ممنوع است.

برای هر مدل این موارد باید ثبت شوند: ID واقعی، تاریخ بررسی، endpoint، JSON/structured output، سقف context و output/reasoning، usage و cache، نرخ و واحد قیمت، timeout و خطا، وضعیت available/unverified/unavailable و علت. `/models/status` فقط آخرین نتیجهٔ ذخیره‌شده را بخواند؛ هر refresh تست پولی اجرا نکند. پاسخ معتبر JSON همیشه در کد هم validate شود.

کلید فقط از `GAPGPT_API_KEY` در محیط پردازش خوانده می‌شود. نه CLI argument، نه config، نه prompt و نه log. اجرای زنده علاوه بر کلید نیازمند `--provider gapgpt --enable-live` و سقف صریح آزمون است؛ fake پیش‌فرض باقی می‌ماند. کلید ارسال‌شده در گفتگو پیش از بهره‌برداری بهتر است تعویض شود.
