def parse_llm_json(content: str) -> Dict[str, Any]:
    """Parse JSON from model response, handling markdown fences, thoughts, control chars, and truncations."""
    if not content or not content.strip():
        raise ValueError("پاسخ مدل خالی بود.")

    text = content.strip()

    # 1. Remove reasoning / thinking tags (<think>...</think>) from thinking models
    text = re.sub(r"<think>[\s\S]*?</think>", "", text).strip()

    # 2. Try direct JSON parse with strict=False (allows unescaped control chars / newlines)
    try:
        res = json.loads(text, strict=False)
        if isinstance(res, dict):
            return res
    except Exception:
        pass

    # 3. Check for markdown code fence ```json ... ``` or ``` ... ```
    fence_match = re.search(r"```(?:json)?\s*([\s\S]*?)\s*```", text)
    if fence_match:
        candidate = fence_match.group(1).strip()
        try:
            res = json.loads(candidate, strict=False)
            if isinstance(res, dict):
                return res
        except Exception:
            pass

    # 4. Use json_repair if available (repairs unclosed strings, missing braces/brackets, bad escapes)
    if json_repair is not None:
        try:
            repaired = json_repair.repair_json(text, return_objects=True)
            if isinstance(repaired, dict):
                return repaired
            if isinstance(repaired, str) and repaired.strip():
                try:
                    res = json.loads(repaired, strict=False)
                    if isinstance(res, dict):
                        return res
                except Exception:
                    pass
        except Exception:
            pass

    # 5. Extract substring between first '{' and last '}'
    start = text.find("{")
    end = text.rfind("}")
    if start != -1 and end != -1 and end > start:
        candidate = text[start : end + 1]
        try:
            res = json.loads(candidate, strict=False)
            if isinstance(res, dict):
                return res
        except Exception:
            if json_repair is not None:
                try:
                    repaired = json_repair.repair_json(candidate, return_objects=True)
                    if isinstance(repaired, dict):
                        return repaired
                except Exception:
                    pass

    # 6. If truncated JSON (e.g. cut off at max tokens), attempt manual closure
    if start != -1:
        candidate = text[start:]
        open_braces = candidate.count("{") - candidate.count("}")
        open_brackets = candidate.count("[") - candidate.count("]")
        if open_braces > 0 or open_brackets > 0:
            repaired = candidate.rstrip()
            if repaired.endswith(","):
                repaired = repaired[:-1]
            repaired += ("]" * max(0, open_brackets)) + ("}" * max(0, open_braces))
            try:
                res = json.loads(repaired, strict=False)
                if isinstance(res, dict):
                    return res
            except Exception:
                pass

    # 7. Final fallback to json.loads to raise standardized JSONDecodeError
    return json.loads(text)

[tests/test_invalid_json.py::test_invalid_json_from_model_handled_and_logged]
@pytest.mark.asyncio
async def test_invalid_json_from_model_handled_and_logged(engine, fake_provider, sample_task_data):
    """When LLM returns corrupted or non-JSON output, engine does not crash unhandled;
    it logs an event, handles the error, and transitions run to FAILED.
    """
    fake_provider.simulate_invalid_json = True

    run_id, _ = engine.submit_run(sample_task_data)
    result = await engine.execute_run(run_id)

    assert result.status == "FAILED"
    assert "Invalid" in result.summary or "JSON" in result.summary

    # Verify event was recorded in DB
    events = engine.repo.get_events(run_id)
    fail_event = next((e for e in events if e["to_state"] == "FAILED"), None)
    assert fail_event is not None
    assert "JSON" in fail_event["reason"]

    def context_priority(path: Path) -> tuple[int, str]:
        relative = path.relative_to(root).as_posix().lower()
        # Code is more useful to a review than generated reports, test cache, or lockfiles.
        if "/src/" in f"/{relative}" or relative.startswith("src/"):
            return (0, relative)
        if relative.startswith(("electron-app/", "app/", "lib/")):
            return (1, relative)
        if path.name.lower() in {"readme.md", "pyproject.toml", "package.json"}:
            return (2, relative)
        if "/tests/" in f"/{relative}" or relative.startswith("tests/"):
            return (4, relative)
        return (3, relative)
