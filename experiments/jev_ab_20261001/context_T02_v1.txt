    def usage(self):
        today = now()[:10]
        summary = dict(self.conn.execute('''SELECT COUNT(*) AS runs,
            COALESCE(SUM(spent_cost_usd),0) AS spent,
            COALESCE(SUM(reserved_cost_usd),0) AS reserved,
            COALESCE(SUM(unknown_cost_usd),0) AS unknown,
            COALESCE(SUM(total_calls),0) AS calls FROM runs WHERE substr(created_at,1,10)=?''', (today,)).fetchone())
        summary["day"] = today
        summary["timezone"] = "UTC"
        summary["cap"] = self.preferences()["daily_budget"]
        summary["remaining"] = max(0, summary["cap"]-summary["spent"]-summary["reserved"]-summary["unknown"])
        summary["models"] = [dict(r) for r in self.conn.execute('''
            SELECT model_id,SUM(input_tokens) AS input_tokens,SUM(output_tokens) AS output_tokens,
            SUM(COALESCE(actual_cost_usd,0)) AS cost,COUNT(*) AS calls
            FROM usage_records WHERE substr(created_at,1,10)=? GROUP BY model_id''', (today,))]
        return summary

[src/flux/engine/orchestrator.py::FluxEngine.submit_run]
    def submit_run(self, task_data, idempotency_key=None):
        if idempotency_key:
            existing = self.repo.get_run_by_idempotency_key(idempotency_key)
            if existing:
                return existing["run_id"], False
        task_data = dict(task_data)
        task_data.setdefault("agent_settings", self.get_agent_settings().model_dump())
        run_id = "run_" + uuid.uuid4().hex[:12]
        task_id = task_data.get("task_id", "task_" + uuid.uuid4().hex[:8])
        created = self.repo.create_run(run_id=run_id, task_id=task_id,
            root_id=task_data.get("root_id", task_id), project_id=task_data.get("project_id", "personal"),
            human_owner=task_data.get("human_owner", "Me"), task_payload=task_data,
            budget_cap_usd=float(task_data.get("budget_cap_usd", .5)), idempotency_key=idempotency_key)
        if not created:
            existing = self.repo.get_run_by_idempotency_key(idempotency_key)
            if existing:
                return existing["run_id"], False
            raise ValueError("Unable to create run")
        return run_id, True
