    def submit_run(self, task_data, idempotency_key=None):
        if idempotency_key:
            existing = self.repo.get_run_by_idempotency_key(idempotency_key)
            if existing:
                return existing["run_id"], False
        task_data = dict(task_data)
        task_data.setdefault("agent_settings", self.get_agent_settings().model_dump())
        run_id = "run_" + uuid.uuid4().hex[:12]
        task_id = task_data.get("task_id", "task_" + uuid.uuid4().hex[:8])
        created = self.repo.create_run(run_id=run_id, task_id=task_id,
            root_id=task_data.get("root_id", task_id), project_id=task_data.get("project_id", "personal"),
            human_owner=task_data.get("human_owner", "Me"), task_payload=task_data,
            budget_cap_usd=float(task_data.get("budget_cap_usd", .5)), idempotency_key=idempotency_key)
        if not created:
            existing = self.repo.get_run_by_idempotency_key(idempotency_key)
            if existing:
                return existing["run_id"], False
            raise ValueError("Unable to create run")
        return run_id, True

    def get_run_by_idempotency_key(self, idempotency_key: str) -> Optional[dict]:
        with self._lock:
            cursor = self.conn.execute(
                "SELECT * FROM runs WHERE idempotency_key = ?", (idempotency_key,)

def cmd_submit(args):
    path = Path(args.file)
    if not path.exists():
        print(f"Error: Task file '{args.file}' not found.", file=sys.stderr)
        sys.exit(1)

    with open(path, "r", encoding="utf-8") as f:
        task_data = json.load(f)

    engine = get_configured_engine(enable_live=args.live)
    run_id, is_new = engine.submit_run(task_data, idempotency_key=args.idempotency_key)
    print(f"Run submitted: {run_id} (New={is_new})")

    if is_new and not args.no_wait:
        print("Executing run...")
        result = asyncio.run(engine.execute_run(run_id))
        print("\n--- Execution Result ---")
        print(f"Status: {result.status}")
        print(f"Summary: {result.summary}")
        print(f"Cost Spent: ${result.spent_cost_usd:.6f}")
        print(f"Total Calls: {result.total_calls}")
        if result.decision:
            print(f"Review Decision: {result.decision.decision} (Confidence={result.decision.confidence})")
    elif not is_new:
        run = engine.repo.get_run(run_id)
        print(f"Existing run found with status: {run['status'] if run else 'UNKNOWN'}")

@pytest.mark.asyncio
async def test_idempotency_prevents_duplicate_runs(engine, fake_provider, sample_task_data):
    """Submitting the same task with the same idempotency key
    returns the existing run_id and is_new=False without executing duplicate calls.
    """
    key = "idem_key_12345"

    run_id_1, is_new_1 = engine.submit_run(sample_task_data, idempotency_key=key)
    assert is_new_1 is True

    # Execute first run
    res_1 = await engine.execute_run(run_id_1)
    assert res_1.status == "ACCEPT_CANDIDATE"
    first_call_count = fake_provider.call_count

    # Submit again with same key
    run_id_2, is_new_2 = engine.submit_run(sample_task_data, idempotency_key=key)
    assert is_new_2 is False
    assert run_id_2 == run_id_1

    # Call count should not have increased
    assert fake_provider.call_count == first_call_count

async def test_invalid_json_from_model_handled_and_logged(engine, fake_provider, sample_task_data):
    """When LLM returns corrupted or non-JSON output, engine does not crash unhandled;
    it logs an event, handles the error, and transitions run to FAILED.
    """
    fake_provider.simulate_invalid_json = True

    run_id, _ = engine.submit_run(sample_task_data)
    result = await engine.execute_run(run_id)

    assert result.status == "FAILED"
    assert "Invalid" in result.summary or "JSON" in result.summary

    # Verify event was recorded in DB
    events = engine.repo.get_events(run_id)
    fail_event = next((e for e in events if e["to_state"] == "FAILED"), None)
    assert fail_event is not None
    assert "JSON" in fail_event["reason"]

[src/flux/storage/repository.py::Repository.create_run]
    def create_run(
        self,
        run_id: str,
        task_id: str,
        root_id: str,
        project_id: str,
        human_owner: str,
        task_payload: dict,
        budget_cap_usd: float,
        idempotency_key: Optional[str] = None
    ) -> bool:
        """Atomically create a run or return False if idempotency conflict."""
        with self._lock:
            now = utc_now_iso()
            try:
                self.conn.execute(
                    """
                    INSERT INTO runs (
                        run_id, task_id, root_id, project_id, human_owner,
                        task_payload, idempotency_key, status, budget_cap_usd,
                        spent_cost_usd, reserved_cost_usd, unknown_cost_usd,
                        total_calls, active_runtime_seconds, created_at, updated_at
                    ) VALUES (?, ?, ?, ?, ?, ?, ?, 'NEW', ?, 0.0, 0.0, 0.0, 0, 0.0, ?, ?)
                    """,
                    (
                        run_id, task_id, root_id, project_id, human_owner,
                        json.dumps(task_payload), idempotency_key, budget_cap_usd,
                        now, now
                    )
                )
                self.conn.commit()
                return True
            except sqlite3.IntegrityError:
                return False
