    @model_validator(mode="after")
    def validate_rules(self) -> "ReviewDecision":
        # Rule: REVISE must provide at least one blocking issue
        if self.decision == "REVISE" and not self.blocking_issues:
            raise ValueError("Decision 'REVISE' requires at least one precise blocking issue.")
        # Rule: ACCEPT_CANDIDATE cannot have blocking issues or failed criteria
        if self.decision == "ACCEPT_CANDIDATE":
            if self.blocking_issues:
                raise ValueError("Decision 'ACCEPT_CANDIDATE' cannot contain blocking issues.")
            if self.failed_criteria:
                raise ValueError("Decision 'ACCEPT_CANDIDATE' cannot contain failed criteria.")
        return self

## ۸. ذخیره‌سازی و API

| Entity | دادهٔ ضروری |
|---|---|
| runs | run_id، task/root/project، قرارداد و hash، state/version، checkpoint، counters، زمان فعال، idempotency و payload hash |
| subtasks | plan معتبر، DAG، role، attempt، lease و نتیجه |
| agent_calls | request hash، model واقعی، prompt version، dispatch state، latency، provider request id، error |
| artifacts | ref، revision، hash، data_class، منشأ و run |
| checks | criterion، وضعیت، evidence refs و ابزار تولیدکننده |
| decisions | review، issues، artifact hash و صادرکننده |
| budget_reservations | root، call، مقدار، price snapshot، وضعیت تسویه/unknown |
| usage_records | توکن‌ها، هزینه تخمینی/واقعی nullable، attempt |
| events | sequence افزایشی، transition، reason، timestamp و metadata پاک‌سازی‌شده |

تغییر state و event متناظر اتمی هستند. فایل artifact ابتدا در فایل موقت نوشته و به‌صورت اتمی نهایی، سپس در DB ارجاع می‌شود؛ recovery فایل بدون رکورد یا ref ناقص را شناسایی کند. migration نسخه‌دار و بازیابی پشتیبان لازم است.

    def _get_client(self) -> AsyncOpenAI:
        key = self._api_key or os.environ.get("GAPGPT_API_KEY", "")
        if not key:
            raise ProviderError(
                "GAPGPT_API_KEY environment variable is not set. Live API calls require GAPGPT_API_KEY.",
                is_transient=False,
                status_code=401
            )
        proxy = os.environ.get("HTTPS_PROXY") or os.environ.get("HTTP_PROXY") or os.environ.get("ALL_PROXY")
        http_client = None
        if proxy:
            import httpx
            http_client = httpx.AsyncClient(proxy=proxy, timeout=120.0)
        return AsyncOpenAI(
            base_url=self.base_url,
            api_key=key,
            http_client=http_client,
            max_retries=0,  # Retries are managed deterministically by Flux engine
            timeout=120.0
        )
